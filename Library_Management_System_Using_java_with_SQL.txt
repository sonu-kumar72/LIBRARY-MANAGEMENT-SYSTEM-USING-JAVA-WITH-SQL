    // ==========================================================
    // CREATE TABLES
    // ==========================================================

    public static void createTables(Connection conn) {

        Statement stmt = null;

        try {

            stmt = conn.createStatement();


            // --------------------------------------------------
            // BOOKS TABLE
            // --------------------------------------------------

            String booksTable =
                    "CREATE TABLE IF NOT EXISTS books (" +
                    "book_id INT PRIMARY KEY AUTO_INCREMENT," +
                    "title VARCHAR(150) NOT NULL," +
                    "author VARCHAR(100) NOT NULL," +
                    "quantity INT NOT NULL DEFAULT 0" +
                    ")";

            stmt.executeUpdate(booksTable);


            // --------------------------------------------------
            // MEMBERS TABLE
            // --------------------------------------------------

            String membersTable =
                    "CREATE TABLE IF NOT EXISTS members (" +
                    "member_id INT PRIMARY KEY AUTO_INCREMENT," +
                    "name VARCHAR(100) NOT NULL," +
                    "phone VARCHAR(20)" +
                    ")";

            stmt.executeUpdate(membersTable);


            // --------------------------------------------------
            // ISSUED BOOKS TABLE
            // --------------------------------------------------

            String issuedTable =
                    "CREATE TABLE IF NOT EXISTS issued_books (" +
                    "issue_id INT PRIMARY KEY AUTO_INCREMENT," +
                    "book_id INT NOT NULL," +
                    "member_id INT NOT NULL," +
                    "issue_date DATE NOT NULL," +
                    "return_date DATE DEFAULT NULL," +
                    "FOREIGN KEY (book_id) " +
                    "REFERENCES books(book_id)," +
                    "FOREIGN KEY (member_id) " +
                    "REFERENCES members(member_id)" +
                    ")";

            stmt.executeUpdate(issuedTable);


            System.out.println(
                    "Database tables are ready."
            );


        } catch (SQLException e) {

            e.printStackTrace();

        } finally {

            try {

                if (stmt != null) {
                    stmt.close();
                }

            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }

    public static void addBook(

        try {

            System.out.println("             ADD BOOK");

            sc.nextLine();


            String sql =
                    "INSERT INTO books "
                    + "(title, author, quantity) "
                    + "VALUES (?, ?, ?)";


            pstmt =
                    conn.prepareStatement(sql);

            pstmt.setString(2, author);
            pstmt.setInt(3, quantity);

        } catch (SQLException e) {

            e.printStackTrace();

        } finally {

            try {

                if (pstmt != null) {
                    pstmt.close();
                }

            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }

    // ==========================================================
    // ISSUE BOOK
    // ==========================================================

    public static void issueBook(
            Connection conn,
            Scanner sc) {

        PreparedStatement checkStmt = null;
        PreparedStatement issueStmt = null;
        PreparedStatement updateStmt = null;
        ResultSet rs = null;

        try {

            System.out.println("\n======================================");
            System.out.println("              ISSUE BOOK");
            System.out.println("======================================");


            System.out.print(
                    "Enter Book ID: "
            );

            int bookId =
                    sc.nextInt();


            System.out.print(
                    "Enter Member ID: "
            );

            int memberId =
                    sc.nextInt();

            sc.nextLine();


            // --------------------------------------------------
            // CHECK BOOK
            // --------------------------------------------------

            String checkBook =
                    "SELECT quantity "
                    + "FROM books "
                    + "WHERE book_id=?";


            checkStmt =
                    conn.prepareStatement(
                            checkBook
                    );


            checkStmt.setInt(
                    1,
                    bookId
            );


            rs =
                    checkStmt.executeQuery();


            if (!rs.next()) {

                System.out.println(
                        "Book ID does not exist."
                );

                return;
            }


            int quantity =
                    rs.getInt("quantity");


            if (quantity <= 0) {

                System.out.println(
                        "Book is not available."
                );

                return;
            }


            // --------------------------------------------------
            // CHECK MEMBER
            // --------------------------------------------------

            String checkMember =
                    "SELECT member_id "
                    + "FROM members "
                    + "WHERE member_id=?";


            PreparedStatement memberStmt =
                    conn.prepareStatement(
                            checkMember
                    );


            memberStmt.setInt(
                    1,
                    memberId
            );


            ResultSet memberRs =
                    memberStmt.executeQuery();


            if (!memberRs.next()) {

                System.out.println(
                        "Member ID does not exist."
                );

                memberRs.close();
                memberStmt.close();

                return;
            }


            memberRs.close();
            memberStmt.close();


            // --------------------------------------------------
            // CHECK DUPLICATE ISSUE
            // --------------------------------------------------

            String duplicateSql =
                    "SELECT issue_id "
                    + "FROM issued_books "
                    + "WHERE book_id=? "
                    + "AND member_id=? "
                    + "AND return_date IS NULL";


            PreparedStatement duplicateStmt =
                    conn.prepareStatement(
                            duplicateSql
                    );


            duplicateStmt.setInt(
                    1,
                    bookId
            );

            duplicateStmt.setInt(
                    2,
                    memberId
            );


            ResultSet duplicateRs =
                    duplicateStmt.executeQuery();


            if (duplicateRs.next()) {

                System.out.println(
                        "This member already has "
                        + "this book."
                );

                duplicateRs.close();
                duplicateStmt.close();

                return;
            }


            duplicateRs.close();
            duplicateStmt.close();


            // --------------------------------------------------
            // INSERT ISSUE RECORD
            // --------------------------------------------------

            String issueSql =
                    "INSERT INTO issued_books "
                    + "(book_id, member_id, issue_date) "
                    + "VALUES (?, ?, CURDATE())";


            issueStmt =
                    conn.prepareStatement(
                            issueSql
                    );


            issueStmt.setInt(
                    1,
                    bookId
            );

            issueStmt.setInt(
                    2,
                    memberId
            );


            issueStmt.executeUpdate();


            // --------------------------------------------------
            // DECREASE BOOK QUANTITY
            // --------------------------------------------------

            String updateSql =
                    "UPDATE books "
                    + "SET quantity = quantity - 1 "
                    + "WHERE book_id=?";


            updateStmt =
                    conn.prepareStatement(
                            updateSql
                    );


            updateStmt.setInt(
                    1,
                    bookId
            );


            updateStmt.executeUpdate();


            System.out.println(
                    "\nBook issued successfully!"
            );


        } catch (SQLException e) {

            e.printStackTrace();

        } finally {

            try {

                if (rs != null) {
                    rs.close();
                }

                if (checkStmt != null) {
                    checkStmt.close();
                }

                if (issueStmt != null) {
                    issueStmt.close();
                }

                if (updateStmt != null) {
                    updateStmt.close();
                }

            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }

            System.out.println("\n======================================");
            System.out.println("              RETURN BOOK");
            System.out.println("======================================");


            System.out.print(
                    "Enter Issue ID: "
            );

            int issueId =
                    sc.nextInt();

            sc.nextLine();


            // --------------------------------------------------
            // FIND ACTIVE ISSUE
            // --------------------------------------------------

            String findSql =
                    "SELECT book_id "
                    + "FROM issued_books "
                    + "WHERE issue_id=? "
                    + "AND return_date IS NULL";


            findStmt =
                    conn.prepareStatement(
                            findSql
                    );


            findStmt.setInt(
                    1,
                    issueId
            );


            rs =
                    findStmt.executeQuery();


            if (!rs.next()) {

                System.out.println(
                        "Active issue record not found."
                );

                return;
            }


            int bookId =
                    rs.getInt("book_id");


            // --------------------------------------------------
            // UPDATE RETURN DATE
            // --------------------------------------------------

            String returnSql =
                    "UPDATE issued_books "
                    + "SET return_date=CURDATE() "
                    + "WHERE issue_id=?";


            returnStmt =
                    conn.prepareStatement(
                            returnSql
                    );


            returnStmt.setInt(
                    1,
                    issueId
            );


            returnStmt.executeUpdate();


            // --------------------------------------------------
            // INCREASE BOOK QUANTITY
            // --------------------------------------------------

            String updateSql =
                    "UPDATE books "
                    + "SET quantity = quantity + 1 "
                    + "WHERE book_id=?";


            updateStmt =
                    conn.prepareStatement(
                            updateSql
                    );


            updateStmt.setInt(
                    1,
                    bookId
            );


            updateStmt.executeUpdate();


            System.out.println(
                    "\nBook returned successfully!"
            );


        } catch (SQLException e) {

            e.printStackTrace();

        } finally {

            try {

                if (rs != null) {
                    rs.close();
                }

                if (findStmt != null) {
                    findStmt.close();
                }

                if (returnStmt != null) {
                    returnStmt.close();
                }

                if (updateStmt != null) {
                    updateStmt.close();
                }

            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }


    // ==========================================================
    // VIEW ISSUED BOOKS
    // ==========================================================

    public static void viewIssuedBooks(
            Connection conn) {

        Statement stmt = null;
        ResultSet rs = null;

        try {

            System.out.println("\n======================================");
            System.out.println("            ISSUED BOOKS");
            System.out.println("======================================");


            stmt =
                    conn.createStatement();


            String sql =
                    "SELECT "
                    + "i.issue_id, "
                    + "b.book_id, "
                    + "b.title, "
                    + "m.member_id, "
                    + "m.name, "
                    + "i.issue_date, "
                    + "i.return_date "
                    + "FROM issued_books i "
                    + "JOIN books b "
                    + "ON i.book_id = b.book_id "
                    + "JOIN members m "
                    + "ON i.member_id = m.member_id "
                    + "ORDER BY i.issue_id";


            rs =
                    stmt.executeQuery(sql);


            System.out.printf(
                    "%-8s %-8s %-25s %-10s %-20s %-15s %-15s%n",
                    "ISSUE ID",
                    "BOOK ID",
                    "BOOK TITLE",
                    "MEMBER ID",
                    "MEMBER NAME",
                    "ISSUE DATE",
                    "RETURN DATE"
            );


            System.out.println(
                    "----------------------------------------------------------------------------------------------------------------"
            );


            boolean found = false;


            while (rs.next()) {

                found = true;


                String returnDate;


                if (rs.getDate("return_date") == null) {

                    returnDate =
                            "Not Returned";

                } else {

                    returnDate =
                            rs.getDate(
                                    "return_date"
                            ).toString();
                }


                System.out.printf(
                        "%-8d %-8d %-25s %-10d %-20s %-15s %-15s%n",
                        rs.getInt("issue_id"),
                        rs.getInt("book_id"),
                        rs.getString("title"),
                        rs.getInt("member_id"),
                        rs.getString("name"),
                        rs.getDate("issue_date"),
                        returnDate
                );
            }


            if (!found) {

                System.out.println(
                        "No issued books found."
                );
            }


        } catch (SQLException e) {

            e.printStackTrace();

        } finally {

            try {

                if (rs != null) {
                    rs.close();
                }

                if (stmt != null) {
                    stmt.close();
                }

            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }
